## 🏆 针对源码的推荐方案

### 为什么源码需要特殊处理？

| 特点         | 普通文档 | 源码                |
| ------------ | -------- | ------------------- |
| 结构化       | ❌       | ✅ (函数、类、模块) |
| 语义复杂     | 低       | 高 (调用关系、继承) |
| 跨文件引用   | 少       | 多 (import、依赖)   |
| 需要精确定位 | ❌       | ✅ (文件、行号)     |

**结论**: 需要 **LSP + 向量搜索 + 关键词** 三者结合！

#### 创建文件

**文件**: `.opencode/tool/code-search.ts`

```typescript
import { Tool } from "@opencode-ai/tool"
import { z } from "zod"
import { execSync } from "child_process"

/**
 * 源码搜索工具（三层搜索）
 */
export const CodeSearchTool = Tool.define("search_source_code", {
  description: "Search source code with multi-layer strategy (LSP + semantic + keyword)",

  parameters: z.object({
    query: z.string().describe("Search query (e.g., 'user authentication implementation')"),
    language: z.enum(["typescript", "javascript", "python", "java", "go"]).optional(),
    searchType: z.enum(["all", "function", "class", "import"]).default("all"),
  }),

  async execute(args, ctx) {
    const results: string[] = []

    // === Layer 1: 关键词快速搜索 ===
    results.push("=== Keyword Search Results ===\n")

    try {
      // 构建文件类型过滤
      const langMap: Record<string, string> = {
        typescript: "ts",
        javascript: "js",
        python: "py",
        java: "java",
        go: "go",
      }

      const typeArg = args.language ? `-t ${langMap[args.language]}` : ""

      // 搜索特定模式
      let pattern = args.query
      if (args.searchType === "function") {
        pattern = `(function|const|def|func).*${args.query}`
      } else if (args.searchType === "class") {
        pattern = `(class|interface|type).*${args.query}`
      } else if (args.searchType === "import") {
        pattern = `(import|require|from).*${args.query}`
      }

      const cmd = `rg -i ${typeArg} -C 2 --max-count 15 "${pattern}"`

      const output = execSync(cmd, {
        cwd: ctx.directory,
        encoding: "utf-8",
        maxBuffer: 1024 * 1024 * 5,
        stdio: ["pipe", "pipe", "ignore"],
      }).toString()

      if (output.trim()) {
        results.push(output)
        results.push("\n")
      } else {
        results.push("No keyword matches found.\n\n")
      }
    } catch (error) {
      results.push("Keyword search failed.\n\n")
    }

    // === Layer 2: 文件路径智能推断 ===
    results.push("=== Smart Path Suggestions ===\n")

    try {
      // 根据查询推断可能的文件路径
      const pathKeywords = args.query.toLowerCase().split(/\s+/)
      const possiblePaths = [
        `**/*${pathKeywords[0]}*.{ts,js,py,java,go}`,
        `**/auth*.{ts,js,py}`, // 如果查询包含 auth
        `**/user*.{ts,js,py}`, // 如果查询包含 user
      ]

      for (const pathPattern of possiblePaths) {
        const files = execSync(`fd -t f "${pathPattern.replace("**", ".")}"`, {
          cwd: ctx.directory,
          encoding: "utf-8",
          stdio: ["pipe", "pipe", "ignore"],
        })
          .toString()
          .split("\n")
          .filter(Boolean)
          .slice(0, 5)

        if (files.length > 0) {
          results.push(`\nRelevant files:\n${files.join("\n")}\n`)
          break
        }
      }
    } catch (error) {
      // Ignore
    }

    // === Layer 3: 使用建议 ===
    results.push("\n=== Next Steps ===\n")
    results.push("💡 Tip: Use LSP tools for precise navigation:\n")
    results.push("  - 'Go to definition of <symbol>'\n")
    results.push("  - 'Find all references of <function>'\n")
    results.push("  - 'Show call hierarchy'\n")

    return {
      title: `Code search results for: "${args.query}"`,
      output: results.join(""),
    }
  },
})
```

#### 步骤 2: 代码解析器

**文件**: `.opencode/knowledge/code-parser.ts`

```typescript
import Parser from "tree-sitter"
import TypeScript from "tree-sitter-typescript"
import fs from "fs"

/**
 * 代码解析器 - 按函数/类分块
 */
export class CodeParser {
  private parser: Parser

  constructor() {
    this.parser = new Parser()
    this.parser.setLanguage(TypeScript.typescript)
  }

  /**
   * 提取函数和类
   */
  extractFunctions(filePath: string): Array<{
    name: string
    type: "function" | "class" | "interface"
    code: string
    startLine: number
    endLine: number
  }> {
    const code = fs.readFileSync(filePath, "utf-8")
    const tree = this.parser.parse(code)
    const functions: any[] = []

    // 遍历 AST
    const traverse = (node: any) => {
      // 函数声明
      if (node.type === "function_declaration" || node.type === "method_definition" || node.type === "arrow_function") {
        const nameNode = node.childForFieldName("name")
        if (nameNode) {
          functions.push({
            name: nameNode.text,
            type: "function",
            code: node.text,
            startLine: node.startPosition.row + 1,
            endLine: node.endPosition.row + 1,
          })
        }
      }

      // 类声明
      if (node.type === "class_declaration") {
        const nameNode = node.childForFieldName("name")
        if (nameNode) {
          functions.push({
            name: nameNode.text,
            type: "class",
            code: node.text,
            startLine: node.startPosition.row + 1,
            endLine: node.endPosition.row + 1,
          })
        }
      }

      // 接口声明
      if (node.type === "interface_declaration") {
        const nameNode = node.childForFieldName("name")
        if (nameNode) {
          functions.push({
            name: nameNode.text,
            type: "interface",
            code: node.text,
            startLine: node.startPosition.row + 1,
            endLine: node.endPosition.row + 1,
          })
        }
      }

      // 递归遍历子节点
      for (let i = 0; i < node.childCount; i++) {
        traverse(node.child(i))
      }
    }

    traverse(tree.rootNode)
    return functions
  }
}
```

#### 步骤 3: 向量数据库（代码优化版）

**文件**: `.opencode/knowledge/code-vector-db.ts`

```typescript
import { ChromaClient } from "chromadb"
import { LocalEmbedding } from "./embedding"
import { CodeParser } from "./code-parser"
import glob from "glob"
import path from "path"

export class CodeKnowledgeBase {
  private client: ChromaClient
  private collection: any
  private embedding: LocalEmbedding
  private parser: CodeParser

  constructor(private projectPath: string) {
    this.client = new ChromaClient()
    this.embedding = new LocalEmbedding()
    this.parser = new CodeParser()
  }

  async init() {
    await this.embedding.init()

    this.collection = await this.client.getOrCreateCollection({
      name: "code_functions",
      metadata: {
        description: "Code knowledge base (function-level)",
        hnsw_space: "cosine", // 余弦相似度
      },
    })

    console.log("Code knowledge base ready!")
  }

  /**
   * 索引代码文件（智能分块）
   */
  async indexCodeFiles(pattern: string = "**/*.{ts,js,tsx,jsx}") {
    const files = glob.sync(pattern, {
      cwd: this.projectPath,
      ignore: ["**/node_modules/**", "**/.git/**", "**/dist/**", "**/build/**"],
    })

    console.log(`Found ${files.length} code files to index`)

    let totalFunctions = 0

    for (let i = 0; i < files.length; i++) {
      const file = files[i]
      const filePath = path.join(this.projectPath, file)

      try {
        // 提取函数和类
        const functions = this.parser.extractFunctions(filePath)

        for (const func of functions) {
          // 构建上下文描述
          const context =
            `File: ${file}\n` +
            `Type: ${func.type}\n` +
            `Name: ${func.name}\n` +
            `Lines: ${func.startLine}-${func.endLine}\n\n` +
            `Code:\n${func.code}`

          // 生成 embedding
          const embedding = await this.embedding.embed(context)

          // 插入数据库
          await this.collection.add({
            ids: [`${file}:${func.name}:${func.startLine}`],
            documents: [context],
            metadatas: [
              {
                file,
                name: func.name,
                type: func.type,
                startLine: func.startLine,
                endLine: func.endLine,
              },
            ],
            embeddings: [embedding],
          })

          totalFunctions++
        }
      } catch (error) {
        console.error(`Error parsing ${file}:`, error.message)
      }

      if ((i + 1) % 10 === 0) {
        console.log(`Indexed ${i + 1}/${files.length} files (${totalFunctions} functions)`)
      }
    }

    console.log(`Indexing complete! Indexed ${totalFunctions} code items.`)
  }

  /**
   * 搜索代码（语义 + 重排序）
   */
  async search(query: string, limit: number = 10) {
    // 1. 向量搜索（语义）
    const queryEmbedding = await this.embedding.embed(query)

    const results = await this.collection.query({
      queryEmbeddings: [queryEmbedding],
      nResults: limit * 2, // 多拿一些候选
    })

    // 2. 结果格式化和重排序
    const formatted = results.documents[0].map((doc: string, i: number) => {
      const meta = results.metadatas[0][i]
      const distance = results.distances[0][i]
      const score = 1 - distance // 转换为相似度

      return {
        file: meta.file,
        name: meta.name,
        type: meta.type,
        startLine: meta.startLine,
        endLine: meta.endLine,
        score,
        snippet: this.extractSnippet(doc, 200),
      }
    })

    // 3. 按分数排序
    formatted.sort((a, b) => b.score - a.score)

    return formatted.slice(0, limit)
  }

  private extractSnippet(code: string, maxLength: number): string {
    const codeStart = code.indexOf("Code:\n") + 6
    const snippet = code.slice(codeStart, codeStart + maxLength)
    return snippet + (snippet.length >= maxLength ? "..." : "")
  }
}
```

#### 步骤 4: MCP Server（完整版）

**文件**: `.opencode/mcp/code-knowledge-server.ts`

```typescript
import { Server } from "@modelcontextprotocol/sdk/server"
import { StdioServerTransport } from "@modelcontextprotocol/sdk/server/stdio"
import { CodeKnowledgeBase } from "../knowledge/code-vector-db"

const server = new Server(
  {
    name: "code-knowledge",
    version: "1.0.0",
  },
  {
    capabilities: { tools: {} },
  },
)

const kb = new CodeKnowledgeBase(process.cwd())
await kb.init()

server.setRequestHandler("tools/list", async () => ({
  tools: [
    {
      name: "search_code_semantic",
      description: "Semantic search in source code (function/class level)",
      inputSchema: {
        type: "object",
        properties: {
          query: {
            type: "string",
            description: "Natural language query (e.g., 'user authentication logic')",
          },
          limit: {
            type: "number",
            default: 5,
            description: "Number of results",
          },
        },
        required: ["query"],
      },
    },
    {
      name: "index_codebase",
      description: "Index source code into knowledge base",
      inputSchema: {
        type: "object",
        properties: {
          pattern: {
            type: "string",
            default: "**/*.{ts,js,tsx,jsx}",
            description: "File pattern to index",
          },
        },
      },
    },
  ],
}))

server.setRequestHandler("tools/call", async (request) => {
  const { name, arguments: args } = request.params

  if (name === "search_code_semantic") {
    const results = await kb.search(args.query, args.limit || 5)

    // 格式化输出
    const formatted = results
      .map((r, i) => {
        return (
          `${i + 1}. [Score: ${r.score.toFixed(3)}] ${r.type} "${r.name}"\n` +
          `   File: ${r.file}:${r.startLine}-${r.endLine}\n` +
          `   Snippet:\n   ${r.snippet}\n`
        )
      })
      .join("\n")

    return {
      content: [
        {
          type: "text",
          text: formatted || "No results found",
        },
      ],
    }
  }

  if (name === "index_codebase") {
    await kb.indexCodeFiles(args.pattern)
    return {
      content: [
        {
          type: "text",
          text: "Code indexing complete!",
        },
      ],
    }
  }
})

const transport = new StdioServerTransport()
await server.connect(transport)
console.error("Code Knowledge MCP Server running!")
```
