# OpenCode 调试日志实战

## 🎯 目标

通过添加调试日志，深入理解 OpenCode CLI 的工作原理。

## 📦 已创建的工具

### 文件位置

`packages/opencode/src/util/debug-log.ts`

### 功能特性

- ✅ 彩色输出（不同模块不同颜色）
- ✅ 精确时间戳
- ✅ 性能计时器
- ✅ 流程追踪器
- ✅ 环境变量控制（不影响生产）

## 🎯 实战：追踪消息处理流程

### 修改 `src/session/processor.ts`

```typescript
import { DebugLog } from "@/util/debug-log"

export namespace SessionProcessor {
  export async function process(sessionID: string, messageID: string) {
    // 创建流程追踪器
    const flow = new DebugLog.FlowTracker("SessionProcessor.process")

    try {
      flow.step("START", { sessionID, messageID })

      // 1. 加载会话
      flow.step("Loading session")
      const session = await Session.get(sessionID)
      DebugLog.object("SESSION", "Session info:", {
        id: session.id,
        agent: session.agent,
        directory: session.directory,
      })

      // 2. 构建上下文
      flow.step("Building context")
      const timer = new DebugLog.Timer("PROMPT", "build")
      const messages = await SessionPrompt.build(sessionID)
      timer.stop()

      DebugLog.session("Context built:", {
        messageCount: messages.length,
        estimatedTokens: estimateTokens(messages),
      })

      // 3. 获取 Provider
      flow.step("Getting provider")
      const provider = await getProvider(session)
      DebugLog.provider(provider.id, "Selected model:", provider.model)

      // 4. 调用 AI
      flow.step("Calling AI")
      const aiTimer = new DebugLog.Timer("AI", "chat")
      const stream = await provider.chat({
        messages,
        tools: getTools(),
        model: provider.model,
      })
      aiTimer.stop()

      // 5. 处理流
      flow.step("Processing stream")
      let chunkCount = 0
      let toolCallCount = 0

      for await (const chunk of stream) {
        chunkCount++

        if (chunk.type === "tool-call") {
          toolCallCount++
          flow.step(`Tool call #${toolCallCount}`, {
            tool: chunk.toolCall.toolName,
          })

          // 执行工具
          const toolTimer = new DebugLog.Timer("TOOL", chunk.toolCall.toolName)
          const result = await executeTool(chunk.toolCall)
          toolTimer.stop()

          DebugLog.tool(chunk.toolCall.toolName, "Result:", {
            success: true,
            outputLength: result.output.length,
          })
        }

        // 每 10 个 chunk 记录一次
        if (chunkCount % 10 === 0) {
          DebugLog.trace("STREAM", "progress", { chunkCount })
        }
      }

      flow.step("Stream complete", { chunkCount, toolCallCount })

      // 6. 更新统计
      flow.step("Updating stats")
      await updateSessionStats(sessionID)

      // 完成
      const stats = flow.complete()

      DebugLog.success("SESSION", "Message processed", {
        totalTime: stats.totalTime,
        steps: stats.steps.length,
      })
    } catch (error) {
      DebugLog.error("SESSION", error as Error, { sessionID, messageID })
      throw error
    }
  }
}
```

## 🔧 实战：追踪工具执行

### 修改 `src/tool/tool.ts`

```typescript
import { DebugLog } from "@/util/debug-log"

export function define<Parameters extends z.ZodType, Result extends Metadata>(
  id: string,
  init: Info<Parameters, Result>["init"],
): Info<Parameters, Result> {
  return {
    id,
    init: async (initCtx) => {
      DebugLog.trace("TOOL", `Initializing ${id}`)

      const toolInfo = init instanceof Function ? await init(initCtx) : init
      const execute = toolInfo.execute

      toolInfo.execute = async (args, ctx) => {
        // 开始追踪
        DebugLog.separator(`Tool Execution: ${id}`)
        DebugLog.tool(id, "Arguments:", args)
        DebugLog.tool(id, "Context:", {
          sessionID: ctx.sessionID,
          messageID: ctx.messageID,
          agent: ctx.agent,
        })

        // 性能计时
        const timer = new DebugLog.Timer("TOOL", id)

        try {
          // 验证参数
          DebugLog.trace("TOOL", `${id}.validate`)
          toolInfo.parameters.parse(args)
          DebugLog.success(`TOOL:${id}`, "Arguments validated")

          // 执行
          DebugLog.trace("TOOL", `${id}.execute`)
          const result = await execute(args, ctx)

          // 记录结果
          const duration = timer.stop()
          DebugLog.tool(id, "Result:", {
            title: result.title,
            outputLength: result.output.length,
            hasAttachments: !!result.attachments?.length,
            duration: `${duration.toFixed(2)}ms`,
          })

          // 截断处理
          const truncated = await Truncate.output(result.output, {}, initCtx?.agent)

          if (truncated.truncated) {
            DebugLog.warn(`TOOL:${id}`, "Output truncated", {
              original: result.output.length,
              truncated: truncated.content.length,
            })
          }

          DebugLog.separator()

          return {
            ...result,
            output: truncated.content,
            metadata: {
              ...result.metadata,
              truncated: truncated.truncated,
              executionTime: duration,
              ...(truncated.truncated && { outputPath: truncated.outputPath }),
            },
          }
        } catch (error) {
          timer.stop("FAILED")
          DebugLog.error(`TOOL:${id}`, error as Error)
          DebugLog.separator()
          throw error
        }
      }

      return toolInfo
    },
  }
}
```

## 🖥️ 实战：追踪 TUI 生命周期

### 修改 `src/cli/cmd/tui/thread.ts`

```typescript
import { DebugLog } from "@/util/debug-log"

export const TuiThreadCommand: CommandModule = {
  // ...
  async handler(opts) {
    const flow = new DebugLog.FlowTracker('TUI Startup')

    try {
      flow.step('Parse options', opts)

      // 确定目录
      const directory = resolveDirectory(opts.project)
      DebugLog.tui('Working directory:', directory)
      flow.step('Directory resolved', { directory })

      // 启动或连接服务器
      if (shouldStartServer(opts)) {
        flow.step('Starting embedded server')
        DebugLog.tui('Starting embedded server...')

        const serverTimer = new DebugLog.Timer('SERVER', 'start')
        const serverUrl = await startServer(directory)
        serverTimer.stop()

        DebugLog.success('SERVER', 'Server started at:', serverUrl)
        flow.step('Server started', { url: serverUrl })
      } else {
        DebugLog.tui('Connecting to remote server:', opts.attach)
        flow.step('Connecting to remote', { url: opts.attach })
      }

      // 创建客户端
      flow.step('Creating SDK client')
      const client = createOpencodeClient({ baseUrl: serverUrl })

      // 订阅事件
      flow.step('Subscribing to events')
      DebugLog.tui('Subscribing to server events...')

      const events = client.global.event()
      let eventCount = 0

      events.on('data', (event) => {
        eventCount++
        if (eventCount % 10 === 0) {
          DebugLog.trace('TUI', 'events received', { count: eventCount })
        }
      })

      // 渲染 UI
      flow.step('Rendering UI')
      DebugLog.tui('Rendering TUI interface...')

      const renderTimer = new DebugLog.Timer('TUI', 'render')
      render(() => <App client={client} />, container)
      renderTimer.stop()

      DebugLog.success('TUI', 'Interface rendered')
      flow.step('UI rendered')

      // 完成
      const stats = flow.complete()

      DebugLog.tui('TUI ready', {
        startupTime: `${stats.totalTime.toFixed(0)}ms`,
        steps: stats.steps.length
      })

    } catch (error) {
      DebugLog.error('TUI', error as Error)
      throw error
    }
  }
}
```

## 💡 调试技巧

### 1. 只启用特定组件的日志

修改 `debug-log.ts`：

```typescript
const enabled = process.env.OPENCODE_DEBUG === "1"
const enabledComponents = (process.env.OPENCODE_DEBUG_COMPONENTS || "").split(",")

function log(emoji: string, color: string, component: string, ...args: any[]) {
  if (!enabled) return
  if (enabledComponents.length > 0 && !enabledComponents.includes(component)) {
    return
  }
  // ... 原有逻辑
}
```

使用：

```bash
# 只看 SESSION 和 TOOL 的日志
export OPENCODE_DEBUG_COMPONENTS=SESSION,TOOL
bun dev
```

### 2. 输出到文件

```bash
# 将日志保存到文件
OPENCODE_DEBUG=1 bun dev 2> debug.log

# 实时查看
tail -f debug.log
```

### 3. 结合 grep 过滤

```bash
# 只看工具执行
OPENCODE_DEBUG=1 bun dev 2>&1 | grep TOOL

# 只看性能数据
OPENCODE_DEBUG=1 bun dev 2>&1 | grep PERF
```
