## 1. 自定义工具开发

### 场景 1.1: 代码质量检查工具

**需求**: 创建一个工具，使用 ESLint/TSC 检查代码质量

```typescript
// packages/opencode/src/tool/code-quality.ts
import { Tool } from "./tool"
import z from "zod"
import { $ } from "bun"
import path from "path"

export const CodeQualityTool = Tool.define("code_quality", {
  description: `Check code quality using linters and type checkers.
    
Supports:
- ESLint for JavaScript/TypeScript
- TypeScript compiler checks
- Prettier format validation
- Custom rules`,

  parameters: z.object({
    files: z.array(z.string()).describe("Files or glob patterns to check"),

    tools: z
      .array(z.enum(["eslint", "tsc", "prettier"]))
      .default(["eslint", "tsc"])
      .describe("Quality check tools to run"),

    fix: z.boolean().default(false).describe("Auto-fix issues when possible"),

    strict: z.boolean().default(false).describe("Fail on warnings (not just errors)"),
  }),

  execute: async (args, ctx) => {
    const results: {
      tool: string
      passed: boolean
      issues: number
      output: string
    }[] = []

    const cwd = Instance.directory

    // ESLint check
    if (args.tools.includes("eslint")) {
      try {
        const eslintArgs = [
          "eslint",
          ...args.files,
          "--format",
          "json",
          ...(args.fix ? ["--fix"] : []),
          ...(args.strict ? ["--max-warnings", "0"] : []),
        ]

        const result = await $`${eslintArgs}`.cwd(cwd).nothrow()
        const eslintResults = JSON.parse(result.stdout.toString())

        const totalIssues = eslintResults.reduce((sum, file) => sum + file.errorCount + file.warningCount, 0)

        results.push({
          tool: "ESLint",
          passed: result.exitCode === 0,
          issues: totalIssues,
          output: formatESLintOutput(eslintResults),
        })
      } catch (error) {
        results.push({
          tool: "ESLint",
          passed: false,
          issues: -1,
          output: `ESLint failed: ${error.message}`,
        })
      }
    }

    // TypeScript check
    if (args.tools.includes("tsc")) {
      try {
        const result = await $`tsc --noEmit ${args.files}`.cwd(cwd).nothrow()

        results.push({
          tool: "TypeScript",
          passed: result.exitCode === 0,
          issues: result.stderr ? countTypeErrors(result.stderr) : 0,
          output: result.stderr.toString(),
        })
      } catch (error) {
        results.push({
          tool: "TypeScript",
          passed: false,
          issues: -1,
          output: `TypeScript check failed: ${error.message}`,
        })
      }
    }

    // Prettier check
    if (args.tools.includes("prettier")) {
      try {
        const result = await $`prettier --check ${args.files}`.cwd(cwd).nothrow()

        results.push({
          tool: "Prettier",
          passed: result.exitCode === 0,
          issues: result.exitCode === 0 ? 0 : 1,
          output: result.stdout.toString(),
        })
      } catch (error) {
        results.push({
          tool: "Prettier",
          passed: false,
          issues: -1,
          output: `Prettier check failed: ${error.message}`,
        })
      }
    }

    // Format output
    const allPassed = results.every((r) => r.passed)
    const totalIssues = results.reduce((sum, r) => sum + Math.max(0, r.issues), 0)

    let output = `# Code Quality Check Results\n\n`
    output += `**Status**: ${allPassed ? "✅ PASSED" : "❌ FAILED"}\n`
    output += `**Total Issues**: ${totalIssues}\n\n`

    for (const result of results) {
      output += `## ${result.tool}\n`
      output += `- Status: ${result.passed ? "✅ Passed" : "❌ Failed"}\n`
      output += `- Issues: ${result.issues >= 0 ? result.issues : "N/A"}\n\n`

      if (result.output) {
        output += `\`\`\`\n${result.output}\n\`\`\`\n\n`
      }
    }

    return {
      title: `Code Quality Check: ${allPassed ? "Passed" : "Failed"}`,
      metadata: {
        allPassed,
        totalIssues,
        results,
      },
      output,
    }
  },
})

function formatESLintOutput(results: any[]): string {
  let output = ""
  for (const file of results) {
    if (file.messages.length > 0) {
      output += `${file.filePath}:\n`
      for (const msg of file.messages) {
        output += `  ${msg.line}:${msg.column} ${msg.severity === 2 ? "error" : "warning"} ${msg.message} (${msg.ruleId})\n`
      }
      output += "\n"
    }
  }
  return output || "No issues found"
}

function countTypeErrors(stderr: Buffer | string): number {
  const text = stderr.toString()
  const matches = text.match(/Found (\d+) errors?/)
  return matches ? parseInt(matches[1]) : 0
}
```

```typescript
// packages/opencode/src/tool/git-enhanced.ts
import { Tool } from "./tool"
import z from "zod"
import { $ } from "bun"

export const GitEnhancedTool = Tool.define("git_enhanced", {
  description: `Enhanced Git operations with AI assistance.
    
Capabilities:
- Smart commit message generation
- Branch management
- Conflict resolution assistance
- Git history analysis`,

  parameters: z.object({
    operation: z.enum(["smart_commit", "create_branch", "resolve_conflict", "analyze_history"]),

    // Smart commit parameters
    files: z.array(z.string()).optional().describe("Files to commit (for smart_commit)"),

    // Branch parameters
    branchName: z.string().optional().describe("Branch name (for create_branch)"),
    fromBranch: z.string().optional().describe("Base branch (for create_branch)"),

    // Conflict parameters
    conflictFile: z.string().optional().describe("File with conflicts (for resolve_conflict)"),

    // History parameters
    since: z.string().optional().describe("Date/commit to analyze from (for analyze_history)"),
    author: z.string().optional().describe("Author filter (for analyze_history)"),
  }),

  execute: async (args, ctx) => {
    const cwd = Instance.directory

    switch (args.operation) {
      case "smart_commit": {
        // 获取变更
        const status = await $`git status --porcelain`.cwd(cwd).text()
        const diff = await $`git diff ${args.files || []}`.cwd(cwd).text()

        // 使用 AI 生成提交信息
        const commitMessage = await generateCommitMessage(diff, status)

        // 执行提交
        if (args.files?.length) {
          await $`git add ${args.files}`.cwd(cwd)
        } else {
          await $`git add .`.cwd(cwd)
        }

        await $`git commit -m ${commitMessage}`.cwd(cwd)

        return {
          title: "Smart Commit Complete",
          metadata: { commitMessage },
          output: `# Commit Created\n\n\`\`\`\n${commitMessage}\n\`\`\`\n\n## Changes\n\`\`\`\n${status}\n\`\`\``,
        }
      }

      case "create_branch": {
        if (!args.branchName) {
          throw new Error("branchName is required for create_branch")
        }

        // 检出基础分支
        if (args.fromBranch) {
          await $`git checkout ${args.fromBranch}`.cwd(cwd)
          await $`git pull`.cwd(cwd)
        }

        // 创建并切换到新分支
        await $`git checkout -b ${args.branchName}`.cwd(cwd)

        return {
          title: `Branch Created: ${args.branchName}`,
          metadata: { branchName: args.branchName },
          output: `Created and switched to branch '${args.branchName}'${args.fromBranch ? ` from '${args.fromBranch}'` : ""}`,
        }
      }

      case "resolve_conflict": {
        if (!args.conflictFile) {
          throw new Error("conflictFile is required for resolve_conflict")
        }

        // 读取冲突文件
        const content = await Bun.file(args.conflictFile).text()

        // 解析冲突
        const conflicts = parseGitConflicts(content)

        // 使用 AI 提供解决建议
        const suggestions = await generateConflictResolution(conflicts)

        return {
          title: `Conflict Analysis: ${args.conflictFile}`,
          metadata: { file: args.conflictFile, conflictCount: conflicts.length },
          output: formatConflictSuggestions(conflicts, suggestions),
        }
      }

      case "analyze_history": {
        const args_arr = [
          "git",
          "log",
          "--pretty=format:%H|%an|%ae|%ad|%s",
          "--date=iso",
          ...(args.since ? [`--since=${args.since}`] : []),
          ...(args.author ? [`--author=${args.author}`] : []),
        ]

        const log = await $`${args_arr}`.cwd(cwd).text()

        // 解析和分析提交历史
        const commits = parseGitLog(log)
        const analysis = analyzeCommits(commits)

        return {
          title: "Git History Analysis",
          metadata: { commitCount: commits.length },
          output: formatHistoryAnalysis(analysis),
        }
      }
    }
  },
})

async function generateCommitMessage(diff: string, status: string): Promise<string> {
  // 使用 AI 生成提交信息
  const { Provider } = await import("../provider/provider")
  const provider = await Provider.get("anthropic")

  const result = await provider.chat({
    model: "claude-3-5-sonnet-20241022",
    messages: [
      {
        role: "user",
        content: `Generate a concise, conventional commit message for these changes:

Status:
${status}

Diff:
${diff.slice(0, 3000)} ${diff.length > 3000 ? "...(truncated)" : ""}

Format: <type>(<scope>): <subject>
Types: feat, fix, docs, style, refactor, test, chore`,
      },
    ],
  })

  return result.choices[0].message.content
}

function parseGitConflicts(content: string) {
  const conflicts = []
  const lines = content.split("\n")
  let inConflict = false
  let currentConflict = null

  for (let i = 0; i < lines.length; i++) {
    const line = lines[i]

    if (line.startsWith("<<<<<<<")) {
      inConflict = true
      currentConflict = {
        start: i,
        ours: [],
        theirs: [],
        base: null,
      }
    } else if (line.startsWith("=======") && inConflict) {
      currentConflict.separator = i
    } else if (line.startsWith(">>>>>>>") && inConflict) {
      currentConflict.end = i
      conflicts.push(currentConflict)
      inConflict = false
      currentConflict = null
    } else if (inConflict) {
      if (!currentConflict.separator || i < currentConflict.separator) {
        currentConflict.ours.push(line)
      } else {
        currentConflict.theirs.push(line)
      }
    }
  }

  return conflicts
}

function parseGitLog(log: string) {
  return log
    .split("\n")
    .filter(Boolean)
    .map((line) => {
      const [hash, author, email, date, ...subjectParts] = line.split("|")
      return {
        hash,
        author,
        email,
        date: new Date(date),
        subject: subjectParts.join("|"),
      }
    })
}

function analyzeCommits(commits: any[]) {
  const byAuthor = new Map<string, number>()
  const byType = new Map<string, number>()
  const byDay = new Map<string, number>()

  for (const commit of commits) {
    // Author stats
    const count = byAuthor.get(commit.author) || 0
    byAuthor.set(commit.author, count + 1)

    // Type stats (from conventional commits)
    const typeMatch = commit.subject.match(/^(\w+)(\(.+\))?:/)
    if (typeMatch) {
      const type = typeMatch[1]
      byType.set(type, (byType.get(type) || 0) + 1)
    }

    // Day stats
    const day = commit.date.toISOString().split("T")[0]
    byDay.set(day, (byDay.get(day) || 0) + 1)
  }

  return {
    total: commits.length,
    byAuthor: Object.fromEntries(byAuthor),
    byType: Object.fromEntries(byType),
    byDay: Object.fromEntries(byDay),
    timeRange: {
      first: commits[commits.length - 1]?.date,
      last: commits[0]?.date,
    },
  }
}

function formatHistoryAnalysis(analysis: any): string {
  let output = `# Git History Analysis\n\n`
  output += `**Total Commits**: ${analysis.total}\n`
  output += `**Time Range**: ${analysis.timeRange.first?.toLocaleDateString()} to ${analysis.timeRange.last?.toLocaleDateString()}\n\n`

  output += `## By Author\n`
  for (const [author, count] of Object.entries(analysis.byAuthor).sort((a, b) => b[1] - a[1])) {
    output += `- ${author}: ${count}\n`
  }

  output += `\n## By Type\n`
  for (const [type, count] of Object.entries(analysis.byType).sort((a, b) => b[1] - a[1])) {
    output += `- ${type}: ${count}\n`
  }

  return output
}

function formatConflictSuggestions(conflicts: any[], suggestions: any): string {

  return `# Conflict Resolution Suggestions\n\n${JSON.stringify(suggestions, null, 2)}`
}
```

```typescript
// packages/opencode/src/server/collaboration.ts
import { Hono } from "hono"
import { describeRoute, validator } from "hono-openapi"
import { z } from "zod"
import { Session } from "../session"
import { Storage } from "../storage"

export const CollaborationRoute = new Hono()

// 协作会话信息
const CollaborationSession = z.object({
  sessionID: z.string(),
  owner: z.string(),
  collaborators: z.array(
    z.object({
      userID: z.string(),
      username: z.string(),
      role: z.enum(["viewer", "editor", "admin"]),
      joinedAt: z.number(),
    }),
  ),
  settings: z.object({
    allowComments: z.boolean().default(true),
    requireApproval: z.boolean().default(false),
  }),
  createdAt: z.number(),
  updatedAt: z.number(),
})

// 创建协作会话
CollaborationRoute.post(
  "/collaboration/session",
  describeRoute({
    summary: "Create collaboration session",
    description: "Enable collaboration on a session",
    operationId: "collaboration.session.create",
  }),
  validator(
    "json",
    z.object({
      sessionID: z.string(),
      settings: CollaborationSession.shape.settings.optional(),
    }),
  ),
  async (c) => {
    const input = c.req.valid("json")
    const userID = c.get("userID") // From auth middleware

    const collab = {
      sessionID: input.sessionID,
      owner: userID,
      collaborators: [],
      settings: input.settings || {
        allowComments: true,
        requireApproval: false,
      },
      createdAt: Date.now(),
      updatedAt: Date.now(),
    }

    await Storage.write(["collaboration", input.sessionID], collab)

    return c.json(collab)
  },
)

// 邀请协作者
CollaborationRoute.post(
  "/collaboration/session/:id/invite",
  describeRoute({
    summary: "Invite collaborator",
    description: "Invite a user to collaborate on a session",
  }),
  validator(
    "json",
    z.object({
      userID: z.string(),
      role: z.enum(["viewer", "editor", "admin"]),
    }),
  ),
  async (c) => {
    const sessionID = c.req.param("id")
    const input = c.req.valid("json")

    const collab = await Storage.read(["collaboration", sessionID])
    if (!collab) {
      return c.json({ error: "Collaboration session not found" }, 404)
    }

    // Add collaborator
    collab.collaborators.push({
      userID: input.userID,
      username: await getUserName(input.userID),
      role: input.role,
      joinedAt: Date.now(),
    })

    collab.updatedAt = Date.now()

    await Storage.write(["collaboration", sessionID], collab)

    // Send notification
    await sendInviteNotification(input.userID, sessionID)

    return c.json(collab)
  },
)

// 实时协作事件流
CollaborationRoute.get(
  "/collaboration/session/:id/events",
  describeRoute({
    summary: "Collaboration events stream",
    description: "Subscribe to real-time collaboration events",
  }),
  async (c) => {
    const sessionID = c.req.param("id")

    return streamSSE(c, async (stream) => {
      const handler = (event: any) => {
        if (event.sessionID === sessionID) {
          stream.writeSSE({
            data: JSON.stringify(event),
          })
        }
      }

      Bus.on("collaboration.event", handler)

      stream.onAbort(() => {
        Bus.off("collaboration.event", handler)
      })
    })
  },
)

// 评论系统
const Comment = z.object({
  id: z.string(),
  sessionID: z.string(),
  messageID: z.string().optional(),
  author: z.string(),
  content: z.string(),
  createdAt: z.number(),
  resolved: z.boolean().default(false),
})

CollaborationRoute.post(
  "/collaboration/comment",
  validator("json", Comment.omit({ id: true, createdAt: true })),
  async (c) => {
    const input = c.req.valid("json")

    const comment = {
      id: ulid(),
      ...input,
      createdAt: Date.now(),
    }

    await Storage.write(["collaboration", "comment", comment.id], comment)

    // Broadcast to collaborators
    Bus.publish("collaboration.event", {
      type: "comment.created",
      data: comment,
    })

    return c.json(comment)
  },
)
```

### 场景 4.1: TUI 中添加代码diff视图

```typescript
// packages/opencode/src/cli/cmd/tui/components/DiffViewer.tsx
import { For, Show, createSignal, createMemo } from 'solid-js'
import { Box, Text } from '@opentui/solid'
import { parseDiff, type DiffLine } from '@/util/diff'

interface DiffViewerProps {
  diff: string
  contextLines?: number
}

export function DiffViewer(props: DiffViewerProps) {
  const [expandedChunks, setExpandedChunks] = createSignal<Set<number>>(new Set())

  const parsed = createMemo(() => parseDiff(props.diff))

  const toggleChunk = (index: number) => {
    setExpandedChunks(prev => {
      const next = new Set(prev)
      if (next.has(index)) {
        next.delete(index)
      } else {
        next.add(index)
      }
      return next
    })
  }

  return (
    <Box flexDirection="column" padding={1}>
      <For each={parsed()}>
        {(file, fileIndex) => (
          <Box flexDirection="column" marginBottom={1}>
            {/* File header */}
            <Box borderStyle="round" borderColor="blue">
              <Text bold color="blue">
                {file.from} → {file.to}
              </Text>
            </Box>

            {/* Hunks */}
            <For each={file.hunks}>
              {(hunk, hunkIndex) => {
                const isExpanded = () => expandedChunks().has(
                  fileIndex() * 1000 + hunkIndex()
                )

                return (
                  <Box flexDirection="column" marginTop={1}>
                    {/* Hunk header */}
                    <Box
                      onClick={() => toggleChunk(fileIndex() * 1000 + hunkIndex())}
                      cursor="pointer"
                    >
                      <Text color="cyan">
                        @@ -{hunk.oldStart},{hunk.oldLines}
                        +{hunk.newStart},{hunk.newLines} @@
                      </Text>
                    </Box>

                    {/* Lines */}
                    <Show when={isExpanded()}>
                      <For each={hunk.lines}>
                        {(line) => (
                          <DiffLine line={line} />
                        )}
                      </For>
                    </Show>
                  </Box>
                )
              }}
            </For>
          </Box>
        )}
      </For>
    </Box>
  )
}

function DiffLine(props: { line: DiffLine }) {
  const color = () => {
    switch (props.line.type) {
      case 'add': return 'green'
      case 'remove': return 'red'
      case 'context': return 'gray'
      default: return 'white'
    }
  }

  const prefix = () => {
    switch (props.line.type) {
      case 'add': return '+ '
      case 'remove': return '- '
      default: return '  '
    }
  }

  const bgColor = () => {
    switch (props.line.type) {
      case 'add': return '#0a3d0a'
      case 'remove': return '#3d0a0a'
      default: return undefined
    }
  }

  return (
    <Box backgroundColor={bgColor()}>
      <Text color={color()}>
        {prefix()}{props.line.content}
      </Text>
    </Box>
```

### 场景 5.1: 实现数据库存储后端

```typescript
// packages/opencode/src/storage/database.ts
import { Database } from "bun:sqlite"
import { z } from "zod"
import path from "path"

export class DatabaseStorage {
  private db: Database

  constructor(dbPath: string) {
    this.db = new Database(dbPath)
    this.initSchema()
  }

  private initSchema() {
    this.db.run(`
      CREATE TABLE IF NOT EXISTS kv_store (
        key TEXT PRIMARY KEY,
        value TEXT NOT NULL,
        created_at INTEGER NOT NULL,
        updated_at INTEGER NOT NULL
      )
    `)

    this.db.run(`
      CREATE INDEX IF NOT EXISTS idx_key_prefix 
      ON kv_store(key)
    `)
  }

  async write<T>(keyPath: string[], data: T): Promise<void> {
    const key = keyPath.join("/")
    const value = JSON.stringify(data)
    const now = Date.now()

    this.db.run(
      `INSERT OR REPLACE INTO kv_store (key, value, created_at, updated_at)
       VALUES (?, ?, ?, ?)`,
      [key, value, now, now],
    )
  }

  async read<T>(keyPath: string[]): Promise<T | undefined> {
    const key = keyPath.join("/")

    const row = this.db.query(`SELECT value FROM kv_store WHERE key = ?`).get(key)

    if (!row) return undefined

    return JSON.parse(row.value as string)
  }

  async list(keyPath: string[]): Promise<string[]> {
    const prefix = keyPath.join("/") + "/"

    const rows = this.db.query(`SELECT key FROM kv_store WHERE key LIKE ?`).all(prefix + "%")

    const keys = new Set<string>()
    for (const row of rows) {
      const key = (row.key as string).slice(prefix.length)
      const firstSegment = key.split("/")[0]
      keys.add(firstSegment)
    }

    return Array.from(keys)
  }

  async remove(keyPath: string[]): Promise<void> {
    const key = keyPath.join("/")

    this.db.run(`DELETE FROM kv_store WHERE key = ? OR key LIKE ?`, [key, key + "/%"])
  }

  // Advanced queries
  async query(options: {
    prefix?: string[]
    where?: Record<string, any>
    limit?: number
    offset?: number
  }): Promise<any[]> {
    let sql = `SELECT key, value FROM kv_store WHERE 1=1`
    const params: any[] = []

    if (options.prefix) {
      sql += ` AND key LIKE ?`
      params.push(options.prefix.join("/") + "%")
    }

    if (options.limit) {
      sql += ` LIMIT ?`
      params.push(options.limit)
    }

    if (options.offset) {
      sql += ` OFFSET ?`
      params.push(options.offset)
    }

    const rows = this.db.query(sql).all(...params)

    return rows.map((row) => ({
      key: row.key,
      value: JSON.parse(row.value),
    }))
  }

  close() {
    this.db.close()
  }
}
```

```typescript
      }
```

```typescript
        },
        body: JSON.stringify(data),

      return response.json()
    },
```

### 场景 7.1: 实现响应缓存

```typescript
// packages/opencode/src/util/cache.ts
import { LRUCache } from "lru-cache"

export class ResponseCache {
  private cache: LRUCache<string, any>

  constructor(
    options: {
      maxSize?: number
      ttl?: number
    } = {},
  ) {
    this.cache = new LRUCache({
      max: options.maxSize || 100,
      ttl: options.ttl || 60000, // 1 minute default
      updateAgeOnGet: true,
    })
  }

  async getOrCompute<T>(key: string, compute: () => Promise<T>, ttl?: number): Promise<T> {
    const cached = this.cache.get(key)
    if (cached !== undefined) {
      return cached as T
    }

    const value = await compute()
    this.cache.set(key, value, { ttl })

    return value
  }

  invalidate(key: string) {
    this.cache.delete(key)
  }

  invalidatePattern(pattern: RegExp) {
    for (const key of this.cache.keys()) {
      if (pattern.test(key)) {
        this.cache.delete(key)
      }
    }
  }

  clear() {
    this.cache.clear()
  }
}

// 全局缓存实例
export const globalCache = new ResponseCache({
  maxSize: 200,
  ttl: 300000, // 5 minutes
})
```

## 8. 安全增强

### 场景 8.1: 实现审计日志

```typescript
// packages/opencode/src/audit/index.ts
import { Storage } from "../storage"
import { ulid } from "ulid"

export namespace Audit {
  export interface Entry {
    id: string
    timestamp: number
    sessionID: string
    userID?: string
    action: string
    resource: string
    result: "success" | "failure"
    detail?: string
    metadata?: Record<string, any>
  }

  export async function log(entry: Omit<Entry, "id" | "timestamp">) {
    const record: Entry = {
      id: ulid(),
      timestamp: Date.now(),
      ...entry,
    }

    await Storage.write(["audit", "log", record.id], record)

    // Also write to append-only log file
    const logLine = JSON.stringify(record) + "\n"
    await Bun.write(await Bun.file(Config.auditLogPath).writer(), logLine)

    return record
  }

  export async function query(filter: {
    sessionID?: string
    action?: string
    startTime?: number
    endTime?: number
    result?: "success" | "failure"
  }) {
    const allLogs = await Storage.list(["audit", "log"])
    const entries: Entry[] = []

    for (const logID of allLogs) {
      const entry = await Storage.read<Entry>(["audit", "log", logID])
      if (!entry) continue

      if (filter.sessionID && entry.sessionID !== filter.sessionID) continue
      if (filter.action && entry.action !== filter.action) continue
      if (filter.result && entry.result !== filter.result) continue
      if (filter.startTime && entry.timestamp < filter.startTime) continue
      if (filter.endTime && entry.timestamp > filter.endTime) continue

      entries.push(entry)
    }

    return entries.sort((a, b) => b.timestamp - a.timestamp)
  }
}

// Middleware to log all tool executions
export function auditMiddleware(toolID: string, execute: Function) {
  return async (args: any, ctx: any) => {
    const startTime = Date.now()
    let result: any
    let error: any

    try {
      result = await execute(args, ctx)

      await Audit.log({
        sessionID: ctx.sessionID,
        action: `tool.execute.${toolID}`,
        resource: toolID,
        result: "success",
        metadata: {
          args,
          duration: Date.now() - startTime,
        },
      })

      return result
    } catch (e) {
      error = e

      await Audit.log({
        sessionID: ctx.sessionID,
        action: `tool.execute.${toolID}`,
        resource: toolID,
        result: "failure",
        detail: e.message,
        metadata: {
          args,
          duration: Date.now() - startTime,
          error: e.stack,
        },
      })

      throw e
    }
  }
}
```
