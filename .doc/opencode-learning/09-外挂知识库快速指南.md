#### 步骤 2: 编写代码

**文件**: `.opencode/tool/knowledge-search.ts`

```typescript
import { Tool } from "@opencode-ai/tool"
import { z } from "zod"
import { execSync } from "child_process"

/**
 * 简单的知识库搜索工具
 * 使用 ripgrep 搜索项目文件
 */
export const KnowledgeSearchTool = Tool.define("search_knowledge", {
  description: "Search the knowledge base (project files and documentation)",

  parameters: z.object({
    query: z.string().describe("Search query"),
    fileTypes: z.array(z.string()).optional().describe("File types to search (e.g., ['md', 'txt'])"),
    maxResults: z.number().default(20).describe("Maximum number of results"),
  }),

  async execute(args, ctx) {
    try {
      // 构建 ripgrep 命令
      let cmd = `rg --ignore-case --context 2 --max-count ${args.maxResults} "${args.query}"`

      // 添加文件类型过滤
      if (args.fileTypes && args.fileTypes.length > 0) {
        const types = args.fileTypes.map((t) => `--type ${t}`).join(" ")
        cmd += ` ${types}`
      }

      // 执行搜索
      const output = execSync(cmd, {
        cwd: ctx.directory,
        encoding: "utf-8",
        maxBuffer: 1024 * 1024 * 5, // 5MB
        // 忽略错误码（找不到结果时 rg 返回 1）
        stdio: ["pipe", "pipe", "ignore"],
      }).toString()

      if (!output.trim()) {
        return {
          title: "No results found",
          output: `No matches found for: "${args.query}"`,
        }
      }

      // 统计结果
      const lines = output.split("\n").filter((l) => l.trim())
      const fileCount = new Set(lines.filter((l) => l.includes(":")).map((l) => l.split(":")[0])).size

      return {
        title: `Found ${fileCount} files with matches`,
        output: output.slice(0, 10000), // 限制输出大小
        metadata: {
          fileCount,
          totalLines: lines.length,
        },
      }
    } catch (error) {
      return {
        title: "Search failed",
        output: `Error: ${error.message}`,
      }
    }
  },
})
```

### 增强版：支持文档目录

**文件**: `.opencode/tool/doc-search.ts`

```typescript
import { Tool } from "@opencode-ai/tool"
import { z } from "zod"
import { execSync } from "child_process"
import path from "path"

export const DocSearchTool = Tool.define("search_docs", {
  description: "Search documentation files (markdown, txt, pdf)",

  parameters: z.object({
    query: z.string().describe("Search query"),
    docPath: z.string().default("docs").describe("Documentation directory path"),
  }),

  async execute(args, ctx) {
    const docsDir = path.join(ctx.directory, args.docPath)

    // 只搜索文档文件
    const cmd = `rg -i -C 3 "${args.query}" --type md --type txt ${docsDir}`

    try {
      const output = execSync(cmd, {
        encoding: "utf-8",
        maxBuffer: 1024 * 1024 * 5,
        stdio: ["pipe", "pipe", "ignore"],
      }).toString()

      return {
        title: `Documentation search results`,
        output: output || `No documentation found for: "${args.query}"`,
      }
    } catch (error) {
      return {
        title: "No matches",
        output: `No results found in ${args.docPath}`,
      }
    }
  },
})
```

**文件**: `.opencode/mcp/vector-db-client.ts`

```typescript
import { Server } from "@modelcontextprotocol/sdk/server"
import { StdioServerTransport } from "@modelcontextprotocol/sdk/server/stdio"

// === 配置你的向量库 API ===
const VECTOR_DB_API = process.env.VECTOR_DB_API || "http://your-vector-db:8080"
const COLLECTION_NAME = process.env.COLLECTION_NAME || "code_knowledge"

// 创建 MCP Server
const server = new Server(
  {
    name: "vector-db-client",
    version: "1.0.0",
  },
  {
    capabilities: {
      tools: {},
    },
  },
)

// 工具列表
server.setRequestHandler("tools/list", async () => {
  return {
    tools: [
      {
        name: "search_vector_db",
        description: "Search the company vector database",
        inputSchema: {
          type: "object",
          properties: {
            query: {
              type: "string",
              description: "Search query",
            },
            limit: {
              type: "number",
              description: "Number of results",
              default: 5,
            },
          },
          required: ["query"],
        },
      },
    ],
  }
})

// 工具实现
server.setRequestHandler("tools/call", async (request) => {
  if (request.params.name === "search_vector_db") {
    const { query, limit = 5 } = request.params.arguments as any

    // 调用你的向量库 API
    const response = await fetch(`${VECTOR_DB_API}/search`, {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        // 如需认证，添加 header
        Authorization: `Bearer ${process.env.VECTOR_DB_TOKEN}`,
      },
      body: JSON.stringify({
        collection: COLLECTION_NAME,
        query,
        limit,
      }),
    })

    const results = await response.json()

    // 格式化结果
    const formatted = results.data
      .map((item: any, i: number) => {
        return `${i + 1}. [Score: ${item.score.toFixed(3)}] ${item.metadata.file}\n${item.text}\n`
      })
      .join("\n")

    return {
      content: [
        {
          type: "text",
          text: formatted || "No results found",
        },
      ],
    }
  }

  throw new Error("Unknown tool")
})

// 启动
const transport = new StdioServerTransport()
await server.connect(transport)
console.error("Vector DB Client MCP Server running!")
```

import { ChromaClient } from "chromadb"
import { LocalEmbedding } from "./embedding"
import fs from "fs"
import path from "path"
import glob from "glob"

export class KnowledgeBase {
  private client: ChromaClient
  private collection: any
  private embedding: LocalEmbedding

  constructor(private projectPath: string) {
    this.client = new ChromaClient()
    this.embedding = new LocalEmbedding()
  }

  async init() {
    await this.embedding.init()

    this.collection = await this.client.getOrCreateCollection({
      name: "code_knowledge",
      metadata: { description: "Code knowledge base" }
    })

    console.log("Knowledge base ready!")
  }

  async indexFiles(pattern: string = "**/*.{md,txt,ts,js,py}") {
    const files = glob.sync(pattern, {
      cwd: this.projectPath,
      ignore: ["**/node_modules/**", "**/.git/**", "**/dist/**"]
    })

    console.log(`Indexing ${files.length} files...`)

    for (let i = 0; i < files.length; i++) {
      const file = files[i]
      const filePath = path.join(this.projectPath, file)
      const content = fs.readFileSync(filePath, "utf-8")

      // 简单分块（每1000字符）
      const chunks = this.chunkText(content, 1000)

      for (let j = 0; j < chunks.length; j++) {
        const chunk = chunks[j]
        const id = `${file}:${j}`

        // 生成 embedding
        const embedding = await this.embedding.embed(chunk)

        // 插入数据库
        await this.collection.add({
          ids: [id],
          documents: [chunk],
          metadatas: [{ file, chunkIndex: j }],
          embeddings: [embedding]
        })
      }

      if ((i + 1) % 10 === 0) {
        console.log(`Indexed ${i + 1}/${files.length} files`)
      }
    }

    console.log("Indexing complete!")
  }

  private chunkText(text: string, size: number): string[] {
    const chunks: string[] = []
    for (let i = 0; i < text.length; i += size) {
      chunks.push(text.slice(i, i + size))
    }
    return chunks
  }

  async search(query: string, limit: number = 5) {
    const queryEmbedding = await this.embedding.embed(query)

    const results = await this.collection.query({
      queryEmbeddings: [queryEmbedding],
      nResults: limit
    })

    return results.documents[0].map((doc: string, i: number) => ({
      content: doc,
      file: results.metadatas[0][i].file,
      score: 1 - results.distances[0][i],  // 转换为相似度
      chunkIndex: results.metadatas[0][i].chunkIndex
    }))
  }
}
EOF
