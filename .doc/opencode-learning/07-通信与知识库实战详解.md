```typescript
export function createOpencodeClient(config?: Config & { directory?: string }) {
  // 1. 禁用超时（流式请求很长）
  if (!config?.fetch) {
    const customFetch: any = (req: any) => {
      req.timeout = false // ⭐ 关键：允许长时间连接
      return fetch(req)
    }
    config = { ...config, fetch: customFetch }
  }

  // 2. 添加项目目录 header
  if (config?.directory) {
    config.headers = {
      ...config.headers,
      "x-opencode-directory": config.directory, // ⭐ 项目隔离
    }
  }

  // 3. 创建基础 HTTP 客户端
  const client = createClient(config) // @hey-api/openapi-ts

  // 4. 返回高级 SDK
  return new OpencodeClient({ client }) // ⭐ 包装类
}
```

```typescript
export namespace Server {
  export const App = lazy(() => {
    const app = new Hono()

    // === 会话 API ===

    // 创建消息 ⭐⭐⭐
    app.post("/session/:id/message", async (c) => {
      const { id } = c.req.param() // sessionID
      const body = await c.req.json()

      // 1. 创建消息记录
      const message = await MessageV2.create({
        sessionID: id,
        role: body.role,
        parts: body.parts,
      })

      // 2. 异步处理（不阻塞响应）⭐
      SessionProcessor.process(id, message.id).catch((err) => {
        console.error("Process error:", err)
      })

      // 3. 立即返回消息
      return c.json(message)
    })

    // 获取会话
    app.get("/session/:id", async (c) => {
      const { id } = c.req.param()
      const session = await Session.get(id)
      return c.json(session)
    })

    // SSE 事件流 ⭐
    app.get("/global/event", async (c) => {
      return streamSSE(c, async (stream) => {
        // 订阅全局事件总线
        for await (const event of GlobalBus.subscribe()) {
          await stream.writeSSE({
            data: JSON.stringify(event),
          })
        }
      })
    })

    return app
  })
}
```

```typescript
export namespace SessionProcessor {
  /**
   * 处理用户消息（核心业务逻辑）
   */
  export async function process(sessionID: string, messageID: string): Promise<void> {
    // === Step 1: 获取会话信息 ===
    const session = await Session.get(sessionID)
    const agent = session.agent

    // === Step 2: 构建提示词上下文 ===
    const messages = await SessionPrompt.build(sessionID)
    // 结果示例:
    // [
    //   { role: "system", content: "You are..." },
    //   { role: "user", content: "Hello" },
    //   { role: "assistant", content: "Hi!" },
    //   { role: "user", content: "新消息" }
    // ]

    // === Step 3: 获取 Provider（Vercel AI SDK）===
    const providerInfo = await Provider.get(agent.model.providerID)
    // providerInfo 包含：
    // - id: "openai"
    // - sdk: OpenAI SDK 实例
    // - languageModel: (modelID) => model

    // === Step 4: 调用 AI ===
    await LLM.run({
      sessionID,
      messageID,
      messages, // 上下文
      agent, // Agent 配置
      provider: providerInfo,
    })
  }
}
```

**LLM.run() 实现**:

**位置**: `packages/opencode/src/session/llm.ts (第 100-250 行)`

```typescript
import { streamText } from "ai" // ⭐ Vercel AI SDK

export namespace LLM {
  export async function run(input: {
    sessionID: string
    messageID: string
    messages: Message[]
    agent: Agent.Info
    provider: Provider.Info
  }): Promise<void> {
    // 1. 获取模型实例
    const model = await provider.languageModel(input.agent.model.modelID)
    // model 类型: LanguageModelV2 (Vercel AI SDK)

    // 2. 获取工具定义
    const tools = await getTools(input.sessionID)

    // 3. 调用 Vercel AI SDK ⭐⭐⭐
    const { textStream, toolCalls, finishReason } = streamText({
      model, // AI 模型
      messages: input.messages, // 上下文
      tools, // 工具定义
      temperature: input.agent.options.temperature,
      maxTokens: input.agent.options.maxTokens,
      // ... 其他参数
    })

    // 4. 处理文本流
    for await (const textDelta of textStream) {
      // 更新消息
      await MessageV2.appendText(input.messageID, textDelta)

      // 推送 SSE 事件
      await GlobalBus.publish({
        type: "session.message.updated",
        data: {
          sessionID: input.sessionID,
          messageID: input.messageID,
          delta: textDelta,
        },
      })
    }

    // 5. 处理工具调用
    for (const toolCall of toolCalls) {
      const result = await Tool.execute(toolCall.toolName, toolCall.args, {
        sessionID: input.sessionID,
        messageID: input.messageID,
        agent: input.agent,
      })

      // 工具结果添加到消息
      await MessageV2.appendToolResult(input.messageID, {
        toolCallID: toolCall.toolCallId,
        result,
      })
    }

    // 6. 如果有工具调用，继续对话
    if (toolCalls.length > 0) {
      // 创建新消息，包含工具结果
      const nextMessage = await MessageV2.create({
        sessionID: input.sessionID,
        role: "assistant",
        parts: [
          /* 工具结果 */
        ],
      })

      // 递归处理
      await run({
        ...input,
        messageID: nextMessage.id,
      })
    }
  }
}
```

```typescript
export namespace Provider {
  // ⭐ Bundled Vercel AI SDK Providers
  const BUNDLED_PROVIDERS: Record<string, (options: any) => SDK> = {
    "@ai-sdk/openai": createOpenAI,
    "@ai-sdk/anthropic": createAnthropic,
    "@ai-sdk/google": createGoogleGenerativeAI,
    "@ai-sdk/mistral": createMistral,
    "@ai-sdk/groq": createGroq,
    // ... 20+ providers
  }

  /**
   * 获取 Provider 实例
   */
  export async function get(providerID: string): Promise<Info> {
    // 1. 获取配置
    const config = await Config.get()
    const providerConfig = config.provider?.[providerID]

    // 2. 获取 API Key
    const auth = await Auth.get(providerID)
    const apiKey = auth?.key || Env.get(`${providerID.toUpperCase()}_API_KEY`)

    // 3. 确定 npm 包
    const npm = providerConfig?.npm || `@ai-sdk/${providerID}`

    // 4. 创建 SDK 实例 ⭐
    const createFn = BUNDLED_PROVIDERS[npm]
    const sdk = createFn({
      apiKey,
      baseURL: providerConfig?.options?.baseURL,
      headers: providerConfig?.options?.headers,
      ...providerConfig?.options,
    })

    // 5. 返回包装后的 Provider
    return {
      id: providerID,
      name: providerConfig?.name || providerID,
      sdk, // Vercel AI SDK 实例
      languageModel: (modelID: string) => {
        return sdk.chat(modelID) // 或 sdk.languageModel(modelID)
      },
    }
  }
}
```

```typescript
import { pipeline } from "@xenova/transformers"

/**
 * 本地 Embedding 服务（使用 Transformers.js）
 * 模型会自动下载到本地缓存
 */
export class LocalEmbedding {
  private model: any

  async init() {
    // 使用轻量级中文/英文双语模型
    console.log("Loading embedding model...")
    this.model = await pipeline(
      "feature-extraction",
      "Xenova/all-MiniLM-L6-v2", // 小模型，适合本地运行
    )
    console.log("Model loaded!")
  }

  /**
   * 将文本转换为向量
   */
  async embed(text: string): Promise<number[]> {
    const output = await this.model(text, {
      pooling: "mean",
      normalize: true,
    })
    return Array.from(output.data)
  }

  /**
   * 批量 embedding
   */
  async embedBatch(texts: string[]): Promise<number[][]> {
    return Promise.all(texts.map((t) => this.embed(t)))
  }
}
```

##### **步骤 3: 创建向量数据库**

**创建文件**: `.opencode/knowledge/vector-db.ts`

```typescript
import { ChromaClient } from "chromadb"
import { LocalEmbedding } from "./embedding"
import fs from "fs"
import path from "path"

export class KnowledgeBase {
  private client: ChromaClient
  private collection: any
  private embedding: LocalEmbedding

  constructor(private projectPath: string) {
    this.client = new ChromaClient()
    this.embedding = new LocalEmbedding()
  }

  async init() {
    await this.embedding.init()

    // 创建或获取集合
    const collectionName = "code_knowledge"
    this.collection = await this.client.getOrCreateCollection({
      name: collectionName,
      metadata: {
        description: "Company internal code knowledge base",
      },
    })

    console.log(`Collection "${collectionName}" ready!`)
  }

  /**
   * 索引代码文件
   */
  async indexCodeFiles(pattern: string = "**/*.{ts,js,py}") {
    const glob = await import("glob")
    const files = glob.sync(pattern, {
      cwd: this.projectPath,
      ignore: ["**/node_modules/**", "**/.git/**"],
    })

    console.log(`Found ${files.length} files to index`)

    // 批量处理
    for (let i = 0; i < files.length; i += 10) {
      const batch = files.slice(i, i + 10)
      await this.indexBatch(batch)
      console.log(`Indexed ${i + batch.length}/${files.length}`)
    }

    console.log("Indexing complete!")
  }

  private async indexBatch(files: string[]) {
    const documents: string[] = []
    const metadatas: any[] = []
    const ids: string[] = []

    for (const file of files) {
      const filePath = path.join(this.projectPath, file)
      const content = fs.readFileSync(filePath, "utf-8")

      // 分块（每 500 行一块）
      const chunks = this.chunkCode(content, 500)

      for (let i = 0; i < chunks.length; i++) {
        const chunk = chunks[i]
        documents.push(chunk)
        metadatas.push({
          file: file,
          chunkIndex: i,
          totalChunks: chunks.length,
        })
        ids.push(`${file}:${i}`)
      }
    }

    // 生成 embeddings
    const embeddings = await this.embedding.embedBatch(documents)

    // 插入数据库
    await this.collection.add({
      ids,
      documents,
      metadatas,
      embeddings,
    })
  }

  /**
   * 代码分块
   */
  private chunkCode(code: string, maxLines: number = 500): string[] {
    const lines = code.split("\n")
    const chunks: string[] = []

    for (let i = 0; i < lines.length; i += maxLines) {
      chunks.push(lines.slice(i, i + maxLines).join("\n"))
    }

    return chunks
  }

  /**
   * 搜索知识库
   */
  async search(query: string, limit: number = 5) {
    // 1. Query embedding
    const queryEmbedding = await this.embedding.embed(query)

    // 2. 向量搜索
    const results = await this.collection.query({
      queryEmbeddings: [queryEmbedding],
      nResults: limit,
    })

    // 3. 格式化结果
    return results.documents[0].map((doc: string, i: number) => ({
      content: doc,
      file: results.metadatas[0][i].file,
      score: results.distances[0][i],
      chunkIndex: results.metadatas[0][i].chunkIndex,
    }))
  }
}
```

```typescript
import { Server } from "@modelcontextprotocol/sdk/server"
import { StdioServerTransport } from "@modelcontextprotocol/sdk/server/stdio"
import { KnowledgeBase } from "../knowledge/vector-db"

// 创建 MCP Server
const server = new Server(
  {
    name: "knowledge-base",
    version: "1.0.0",
  },
  {
    capabilities: {
      tools: {},
    },
  },

const kb = new KnowledgeBase(process.cwd())

// 初始化知识库
await kb.init()

// 定义工具: 搜索知识库
server.setRequestHandler("tools/list", async () => {
  return {
    tools: [
      {
        name: "search_codebase",
        description: "Search the company internal codebase knowledge base",
        inputSchema: {
          type: "object",
          properties: {
            query: {
              type: "string",
              description: "Search query",
            },
            limit: {
              type: "number",
              description: "Number of results",
              default: 5,
            },
          },
          required: ["query"],
        },
      },
      {
        name: "index_codebase",
        description: "Index code files into the knowledge base",
        inputSchema: {
          type: "object",
          properties: {
            pattern: {
              type: "string",
              description: "File pattern (e.g., '**/*.ts')",
              default: "**/*.{ts,js,py}",
            },
          },
        },
      },
    ],
  }
})

// 实现工具调用
server.setRequestHandler("tools/call", async (request) => {
  const { name, arguments: args } = request.params

  if (name === "search_codebase") {
    const results = await kb.search(args.query, args.limit || 5)

    return {
      content: [
        {
          type: "text",
          text: JSON.stringify(results, null, 2),
        },
      ],
    }
  }

  if (name === "index_codebase") {
    await kb.indexCodeFiles(args.pattern)

    return {
      content: [
        {
          type: "text",
          text: "Indexing complete!",
        },
      ],
    }
  }

  throw new Error(`Unknown tool: ${name}`)
})

// 启动 Server
const transport = new StdioServerTransport()
await server.connect(transport)

console.log("Knowledge Base MCP Server running!")
```

### 2.4 方案 B: 简化版（基于关键词）

如果不想部署 embedding 模型，可以用简化版：

**创建文件**: `.opencode/tool/code-search.ts`

```typescript
import { Tool } from "@opencode-ai/tool"
import { z } from "zod"
import { execSync } from "child_process"

export const CodeSearchTool = Tool.define("search_internal_code", {
  description: "Search company internal codebase using ripgrep",
  parameters: z.object({
    query: z.string().describe("Search query"),
    filePattern: z.string().optional().describe("File pattern (e.g., '*.ts')"),
    contextLines: z.number().default(3).describe("Context lines"),
  }),

  async execute(args, ctx) {
    // 使用 ripgrep 搜索
    const cmd = `rg -i -C ${args.contextLines} "${args.query}" ${
      args.filePattern ? `--glob "${args.filePattern}"` : ""
    }`

    const output = execSync(cmd, {
      cwd: ctx.directory,
      encoding: "utf-8",
      maxBuffer: 1024 * 1024 * 10, // 10MB
    })

    return {
      title: `Found ${output.split("\n").length} lines`,
      output,
    }
  },
})
```
